```python
# news_fetch.py
import hashlib
def dedup_by_hash(items):
    seen, out = set(), []
    for it in items:
        h = hashlib.md5((it["title"] + (it.get("content") or "")).encode()).hexdigest()
        if h in seen: continue
        seen.add(h); out.append(it)
    return out

def fetch_daily_news():
    import akshare as ak
    rows = ak.stock_news_em(symbol="全部")          # 全部A股要闻;接口变更时替换
    return [{"title": r.get("新闻标题", ""), "content": r.get("新闻内容", ""),
             "source": "em"} for r in rows.to_dict("records")[:200]]
```
